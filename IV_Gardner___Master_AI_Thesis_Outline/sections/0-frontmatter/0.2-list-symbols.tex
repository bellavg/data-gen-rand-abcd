\chapter{List of Symbols}
\label{sec:list-symbols}

\begin{table}[h]
\centering
\begin{tabular}{@{} l p{0.68\textwidth} @{}}
\toprule
\textbf{Symbol} & \textbf{Description} \\
\midrule
\multicolumn{2}{@{}l}{\textbf{Graph and structural notation}} \\
\midrule
$G = (\mathcal{V}, \mathcal{E})$ & And-Inverter Graph (AIG): a directed acyclic graph over the basis $\{\wedge, \neg\}$, with node (gate) set $\mathcal{V}$ and edge set $\mathcal{E}$ (\ref{sec:prelim:aig}). \\
$\tau$ & Node-role labelling, $\tau \colon \mathcal{V} \rightarrow \{\mathrm{const}, \mathrm{PI}, \mathrm{AND}, \mathrm{PO}\}$ \eqref{eq:prelim:role}. \\
$\mathrm{inv}$ & Edge-inversion flag, $\mathrm{inv} \colon \mathcal{E} \rightarrow \{0, 1\}$, marking a connection as regular or inverted \eqref{eq:prelim:inversion}. \\
$\operatorname{fi}(v)$, $\operatorname{fo}(v)$ & Fanin set and fanout set of node $v$ \eqref{eq:prelim:fanin}. \\
$\ell(v)$, $\ell(G)$ & Topological level of node $v$, and depth of $G$: $\ell(G) = \max_{v \in \mathcal{V}} \ell(v)$ \eqref{eq:prelim:level}. \\
$\mathcal{C}(v)$, $\mathcal{F}(v)$ & Fanin (causal) cone and fanout cone of node $v$ \eqref{eq:prelim:cone}. \\
$\operatorname{MFFC}(v)$ & Maximal Fanout-Free Cone of $v$: the gates that feed $v$ exclusively, removed with $v$ under restructuring \eqref{eq:prelim:mffc}. \\
\addlinespace
\multicolumn{2}{@{}l}{\textbf{Task, model, and reduction operators}} \\
\midrule
$S$ & Deterministic synthesis algorithm (Orchestrate); $S(G)$ is the function-equivalent, optimized circuit (\ref{sec:prelim:algorithms:orchestrate}). \\
$y(G)$ & Optimizability of $G$ under $S$: the fraction of nodes $S$ removes, $y(G) = 1 - |\mathcal{V}(S(G))| / |\mathcal{V}(G)| \in [0, 1]$ \eqref{eq:prelim:target}. \\
$f_\theta$ & Trained GNN encoder, approximating $y$ from an input AIG (\ref{sec:prelim:formalization}). \\
$R$ & Reduction operator; $R(G)$ is the reduced graph, and $R = \operatorname{id}$ denotes the unreduced baseline (\ref{sec:prelim:reduction}). \\
\addlinespace
\multicolumn{2}{@{}l}{\textbf{Measuring reduction}} \\
\midrule
$\eta_{\mathcal{V}}(G)$, $\eta_{\mathcal{E}}(G)$ & Node and edge retention (compression) ratio of a reduction: $|\mathcal{V}(R(G))| / |\mathcal{V}(G)|$ and $|\mathcal{E}(R(G))| / |\mathcal{E}(G)|$ \eqref{eq:prelim:retention}. \\
$n^{\star}$ & Amortisation threshold: the number of epochs beyond which applying a reduction is cheaper than training on the unreduced graph \eqref{eq:method:amortisation}. \\
\bottomrule
\end{tabular}
\end{table}
